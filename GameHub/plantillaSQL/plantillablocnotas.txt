-- Primer paso crear la base de datos(Este es opcional porque tambien lo podes hacer manualmente y mas cómodo)
CREATE DATABASE IF NOT EXISTS gamehub
CHARACTER SET utf8mb4
COLLATE utf8mb4_general_ci;

-- Despues la seleccionas
USE gamehub;

-- ==========================================
-- Aca creamos la tabla de usuarios
-- ==========================================

DROP TABLE IF EXISTS usuarios;

CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    correo VARCHAR(150) NOT NULL UNIQUE,
    contrasena VARCHAR(255) NOT NULL,
    fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    foto_perfil VARCHAR(255)
);

-- ==========================================
-- Tambien la tabla de juegos
-- ==========================================

DROP TABLE IF EXISTS juegos;

CREATE TABLE juegos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    genero VARCHAR(50) NOT NULL,
    descripcion TEXT,
    imagen VARCHAR(255),
    descripcion_completa TEXT,
    anio_lanzamiento INT,
    curiosidad TEXT
);
